```python
"""Consolidated AP/AR queries: both sides of the ledger in one call."""
import uuid
from datetime import date as _date
from decimal import Decimal
from typing import Optional

from sqlalchemy.ext.asyncio import AsyncSession

from app.models.invoice import Invoice
from app.schemas.ar_ap import AgingSide, ArApOverview, UpcomingInvoice
from app.services.invoice_service import aging_summary

ZERO = Decimal("0")


def _side_to_schema(raw: dict) -> AgingSide:
    upcoming = [
        UpcomingInvoice(
            id=inv.id,
            number=inv.number,
            series=inv.series,
            due_date=inv.due_date.isoformat() if inv.due_date else "",
            total=inv.total or ZERO,
            balance=raw["_balance"](inv),
            payee_name=inv.payee.name if inv.payee else None,
        )
        for inv in raw["upcoming"]
    ]
    return AgingSide(
        outstanding=raw["outstanding"],
        overdue_amount=raw["overdue_amount"],
        overdue_count=raw["overdue_count"],
        current=raw["buckets"]["current"],
        d1_30=raw["buckets"]["d1_30"],
        d31_60=raw["buckets"]["d31_60"],
        d61_90=raw["buckets"]["d61_90"],
        d90_plus=raw["buckets"]["d90_plus"],
        collected_this_month=raw["collected_this_month"],
        upcoming=upcoming,
    )


async def get_overview(
    session: AsyncSession,
    workspace_id: uuid.UUID,
    today: Optional[_date] = None,
) -> ArApOverview:
    """Return aging data for both directions in a single call."""
    from app.services.invoice_service import aging_summary, balance as _balance

    raw_r = await aging_summary(session, workspace_id, today=today, direction="receivable")
    raw_p = await aging_summary(session, workspace_id, today=today, direction="payable")

    # aging_summary does not include a balance callable — attach one so _side_to_schema works
    raw_r["_balance"] = _balance
    raw_p["_balance"] = _balance

    receivable = _side_to_schema(raw_r)
    payable = _side_to_schema(raw_p)

    return ArApOverview(
        receivable=receivable,
        payable=payable,
        net_position=receivable.outstanding - payable.outstanding,
    )
```

from sqlalchemy.ext.asyncio import AsyncSession

from app.core.database import get_async_session
from app.core.permissions import Permission
from app.core.workspace_context import WorkspaceContext, current_workspace
from app.schemas.ar_ap import ArApOverview
from app.services import ar_ap_service

router = APIRouter(prefix="/api/ar-ap", tags=["ar-ap"])


def _read_ctx(ctx: WorkspaceContext = Depends(current_workspace)) -> WorkspaceContext:
    ctx.require_permission(Permission.EXPORT_READ)
    return ctx


@router.get("/overview", response_model=ArApOverview)
async def get_overview(
    ctx: WorkspaceContext = Depends(_read_ctx),
    session: AsyncSession = Depends(get_async_session),
):
    return await ar_ap_service.get_overview(session, ctx.workspace.id)

```python
from app.schemas.dre import DreLine, DreReport
from app.services._query_filters import counts_as_pnl

async def get_dre(
    session: AsyncSession,
    workspace_id: uuid.UUID,
    user_id: uuid.UUID,
    year: int,
    month: int,
    include_prev: bool = True,
) -> DreReport:
    """DRE por competência para o mês indicado.

    Usa `Transaction.date` (competência), não `effective_date` (caixa de cartão).
    Exclui transferências e saldos iniciais via `counts_as_pnl`.
    """
    from datetime import date
    import calendar

    start = date(year, month, 1)
    _, last_day = calendar.monthrange(year, month)
    end = date(year, month, last_day)

    workspace = await session.get(Workspace, workspace_id)
    currency = workspace.primary_currency

    # Busca transações do período que contam como P&L
    result = await session.execute(
        select(Transaction, Category)
        .join(Category, Transaction.category_id == Category.id, isouter=True)
        .where(
            Transaction.workspace_id == workspace_id,
            Transaction.date >= start,
            Transaction.date <= end,
            Transaction.status != "ignored",
            counts_as_pnl(Transaction, user_id),
        )
        .options(selectinload(Transaction.category))
    )
    rows = result.unique().all()

    revenues: dict[str, Decimal] = {}
    expenses: dict[str, Decimal] = {}

    for tx, cat in rows:
        cat_name = cat.name if cat else "Sem categoria"
        # Usar amount_primary se disponível, senão amount
        amount = tx.amount_primary if tx.amount_primary is not None else tx.amount
        if tx.type == "credit":
            revenues[cat_name] = revenues.get(cat_name, ZERO) + amount
        else:
            expenses[cat_name] = expenses.get(cat_name, ZERO) + amount

    gross_revenue = sum(revenues.values(), ZERO)
    operating_expenses = sum(expenses.values(), ZERO)

    lines = (
        [DreLine(label=k, amount=v) for k, v in revenues.items()] +
        [DreLine(label=k, amount=-v) for k, v in expenses.items()]
    )

    report = DreReport(
        year=year,
        month=month,
        currency=currency,
        gross_revenue=gross_revenue,
        deductions=ZERO,
        net_revenue=gross_revenue,
        operating_expenses=operating_expenses,
        result=gross_revenue - operating_expenses,
        lines=lines,
    )

    if include_prev:
        prev_month = month - 1
        prev_year = year
        if prev_month == 0:
            prev_month = 12
            prev_year -= 1
        report.prev_month = await get_dre(
            session, workspace_id, user_id, prev_year, prev_month, include_prev=False
        )

    return report
```

from app.schemas.dre import DfcPeriod, DfcReport

async def get_dfc(
    session: AsyncSession,
    workspace_id: uuid.UUID,
    user_id: uuid.UUID,
    year: int,
    month: int,
) -> DfcReport:
    """DFC direto (caixa): entradas e saídas reais por semana no mês."""
    import calendar
    from datetime import date, timedelta

    start = date(year, month, 1)
    _, last_day = calendar.monthrange(year, month)
    end = date(year, month, last_day)

    workspace = await session.get(Workspace, workspace_id)
    currency = workspace.primary_currency

    result = await session.execute(
        select(Transaction)
        .where(
            Transaction.workspace_id == workspace_id,
            Transaction.date >= start,
            Transaction.date <= end,
            Transaction.status != "ignored",
            counts_as_pnl(Transaction, user_id),
        )
    )
    transactions = result.scalars().all()

    # Agrupa por semana (segunda-feira)
    weekly: dict[date, dict[str, Decimal]] = {}
    for tx in transactions:
        week_start = tx.date - timedelta(days=tx.date.weekday())
        if week_start not in weekly:
            weekly[week_start] = {"inflows": ZERO, "outflows": ZERO}
        amount = tx.amount_primary if tx.amount_primary is not None else tx.amount
        if tx.type == "credit":
            weekly[week_start]["inflows"] += amount
        else:
            weekly[week_start]["outflows"] += amount

    # Saldo de abertura = soma das transações anteriores ao mês
    opening_result = await session.execute(
        select(func.coalesce(
            func.sum(
                case(
                    (Transaction.type == "credit", Transaction.amount),
                    else_=-Transaction.amount
                )
            ),
            Decimal("0")
        )).where(
            Transaction.workspace_id == workspace_id,
            Transaction.date < start,
            Transaction.status != "ignored",
            counts_as_pnl(Transaction, user_id),
        )
    )
    opening_balance = opening_result.scalar() or ZERO

    periods = []
    running = opening_balance
    for week_start in sorted(weekly):
        w = weekly[week_start]
        running += w["inflows"] - w["outflows"]
        periods.append(DfcPeriod(
            date=week_start.isoformat(),
            inflows=w["inflows"],
            outflows=w["outflows"],
            balance=running,
        ))

    closing = opening_balance + sum(
        (w["inflows"] - w["outflows"] for w in weekly.values()), ZERO
    )

    return DfcReport(
        year=year,
        month=month,
        currency=currency,
        opening_balance=opening_balance,
        closing_balance=closing,
        periods=periods,
    )
```

async def get_dfc(
    year: int = Query(..., ge=2000, le=2100),
    month: int = Query(..., ge=1, le=12),
    ctx: WorkspaceContext = Depends(current_workspace),
    session: AsyncSession = Depends(get_async_session),

@router.get("/dre/export")
async def export_dre(
    year: int = Query(..., ge=2000, le=2100),
    month: int = Query(..., ge=1, le=12),
    fmt: str = Query("xlsx", alias="format", pattern="^xlsx$"),
    ctx: WorkspaceContext = Depends(current_workspace),
    session: AsyncSession = Depends(get_async_session),
):

    wb = openpyxl.Workbook()
    ws = wb.active
    ws.title = f"DRE {month:02d}/{year}"
    ws.append(["Demonstrativo de Resultado — " + f"{month:02d}/{year}"])
    ws.append([])
    ws.append(["Receita Bruta", float(report.gross_revenue)])
    ws.append(["Deduções", float(report.deductions)])
    ws.append(["Receita Líquida", float(report.net_revenue)])
    ws.append(["Despesas Operacionais", float(report.operating_expenses)])
    ws.append(["Resultado", float(report.result)])
    ws.append([])
    ws.append(["Detalhamento"])
    for line in report.lines:
        ws.append(["  " + line.label, float(line.amount)])

    buf = BytesIO()
    wb.save(buf)
    buf.seek(0)

    filename = f"dre_{year}_{month:02d}.xlsx"

        <p className="text-muted-foreground">{t('dre.noData')}</p>
      )}

      <div>
        <h3 className="font-semibold mb-3">{t('dre.dfc')}</h3>
        {loadingDfc ? (
          <Skeleton className="h-48" />
        ) : dfc && dfc.periods.length > 0 ? (
          <div className="rounded-xl border bg-card p-4">
            <ResponsiveContainer width="100%" height={200}>
              <BarChart data={dfc.periods}>
                <CartesianGrid strokeDasharray="3 3" />
                <XAxis dataKey="date" tick={{ fontSize: 11 }} />
                <YAxis tick={{ fontSize: 11 }} />
                <Tooltip formatter={(v: number) => fmt(v)} />
                <Bar dataKey="inflows" name={t('dre.inflows')} fill="#22c55e" />
                <Bar dataKey="outflows" name={t('dre.outflows')} fill="#ef4444" />
              </BarChart>
            </ResponsiveContainer>
          </div>
        ) : (
          <p className="text-muted-foreground">{t('dre.noData')}</p>
        )}
      </div>
    </div>
  )
}
```
